   262→  async function loadTodayStats() {
   263→    if (!user) return;

   265→      const res = await ApiService.getRecoverySummary(getTodayDateStr(), selectedCompanyId || undefined);

   273→        // Merge API visited with locally tracked visited (local takes precedence)
   274→        setVisitedShopIds((prev) => {
   275→          const merged = new Set([...prev, ...visited]);
   276→          StorageService.saveVisitedShops([...merged]);

   278→        });

   282→        setCompanyRecoveryBreakdown(res.companyBreakdown);
   283→      }
   284→
   285→      // ── Sync recoverySubmittedShopIds with server ──────────────────────────
   286→      // If admin rejected a recovery, remove that shop+company from the submitted set
   287→      // so the orderbooker can re-add recovery for that shop under that company
   288→      if (recoverySubmittedShopIds.size > 0) {
   289→        try {
   290→          const todayTxns = await ApiService.getTransactions({
   291→            createdBy: user.id,
   292→            type: 'recovery',
   293→            date: getTodayDateStr(),
   294→            limit: 500,
   295→          });
   296→          // For each recovery key in recoverySubmittedShopIds, check if ALL today's recoveries for that
   297→          // shopId+companyId combo are rejected
   298→          const keysToRemove: string[] = [];
   299→          for (const key of recoverySubmittedShopIds) {
   300→            // Parse key: could be "shopId" or "shopId_companyId"
   301→            const lastUnderscore = key.lastIndexOf('_');
   302→            // Check if the part after last underscore looks like a companyId (cuid format: 25+ chars)
   303→            let shopId: string;
   304→            let companyId: string | undefined;
   305→            if (lastUnderscore > 0 && key.length - lastUnderscore - 1 > 20) {
   306→              // Likely a "shopId_companyId" composite key
   307→              shopId = key.substring(0, lastUnderscore);
   308→              companyId = key.substring(lastUnderscore + 1);
   309→            } else {
   310→              // Legacy "shopId" only key
   311→              shopId = key;
   312→            }
   313→            // Filter transactions for this shopId AND companyId (if applicable)
   314→            const shopTxns = todayTxns.transactions.filter((t) => {
   315→              if (t.shopId !== shopId) return false;
   316→              if (companyId && t.companyId && t.companyId !== companyId) return false;
   317→              return true;
   318→            });
   319→            // Only remove if we found transactions AND all of them are rejected
   320→            // (if no transactions found, don't remove — might be offline/not synced yet)
   321→            if (shopTxns.length > 0 && shopTxns.every((t) => t.status === 'rejected')) {
   322→              keysToRemove.push(key);
   323→            }
   324→          }
   325→          if (keysToRemove.length > 0) {
   326→            console.log('[loadTodayStats] Removing shop+company keys with only rejected recoveries:', keysToRemove);
   327→            setRecoverySubmittedShopIds((prev) => {
   328→              const next = new Set(prev);
   329→              for (const id of keysToRemove) {
   330→                next.delete(id);
   331→                StorageService.removeRecoverySubmittedShop(id);
   332→              }
   333→              return next;
   334→            });
   335→          }
   336→        } catch (e) {
   337→          console.warn('[loadTodayStats] Failed to sync recovery rejection statuses:', e);
   338→        }
   339→      }
   340→      // If myEntry not found, don't reset — keep cached value
   341→    } catch {
   342→      // API failed — keep cached value, don't reset to 0
   343→      console.warn('[loadTodayStats] Failed to fetch, using cached value');
   344→    }
   345→  }

   494→            shopName,
   495→            shopAddress: recoveryShop.address || recoveryShop.area || undefined,
   496→            shopOwnerName: recoveryShop.ownerName || undefined,
   497→            shopPhone,
   498→            openingBalance,
   499→            recoveryAmount: payload.amount,
   500→            remainingBalance,
   501→            companyName: selectedCompanyName || undefined,
   502→            orderbookerName: user.name || undefined,
   503→            distributorPhone: distributorPhone || undefined,
   504→          });
   505→
   506→          // Fallback timeout in case SuccessOverlay doesn't dismiss properly
   507→          // This ensures the SMS popup always shows even if overlay has issues
   508→          setTimeout(() => {
   509→            setNotifChoice((prev) => {

   511→              if (prev.visible) return prev;
   512→              return {
   513→                visible: true,
