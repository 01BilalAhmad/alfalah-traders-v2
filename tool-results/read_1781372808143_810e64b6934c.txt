    44→    44→// Error Boundary class component — catches React render errors that try-catch cannot
    45→    45→class MapErrorBoundary extends Component<
    46→    46→  { children: React.ReactNode; onError: () => void },
    47→    47→  { hasError: boolean; error: Error | null }
    48→    48→> {
    49→    49→  state = { hasError: false, error: null as Error | null };
    50→    50→
    51→    51→  static getDerivedStateFromError(error: Error) {
    52→    52→    return { hasError: true, error };
    53→    53→  }
    54→    54→
    55→    55→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    56→    56→    console.error('[Map] ErrorBoundary caught:', error, info);
    57→    57→    // Notify parent so it can show fallback
    58→    58→    this.props.onError();
    59→    59→  }
    60→    60→
    61→    61→  render() {
    62→    62→    if (this.state.hasError) {
    63→    63→      return null; // Parent will show fallback
    64→    64→    }
    65→    65→    return this.props.children;
    66→    66→  }
    67→    67→}

   169→   169→function MapScreenInner() {
   170→   170→  const insets = useSafeAreaInsets();
   171→   171→  const { user, selectedCompanyId } = useAuth();
   172→   172→  const { todayShops, isOnline } = useShops();
   173→   173→  const { isTracking, startTime } = useRouteTracking();
   174→   174→
   175→   175→  const mapRef = useRef<any>(null);
   176→   176→  const [currentLocation, setCurrentLocation] = useState<{ lat: number; lng: number } | null>(null);
   177→   177→  const [routePoints, setRoutePoints] = useState<Array<{ latitude: number; longitude: number }>>([]);
   178→   178→  const [selectedShop, setSelectedShop] = useState<Shop | null>(null);
   179→   179→  const [isLoadingLocation, setIsLoadingLocation] = useState(true);
   180→   180→  const [mapType, setMapType] = useState<'standard' | 'satellite'>('standard');
   181→   181→  const [showVisitedOnly, setShowVisitedOnly] = useState(false);
   182→   182→  const [visitedShopIds, setVisitedShopIds] = useState<Set<string>>(new Set());
   183→   183→
   184→   184→  // Load current location
   185→   185→  useEffect(() => {
   186→   186→    (async () => {
   187→   187→      try {
   188→   188→        const Location = require('expo-location');
   189→   189→        const { status } = await Location.requestForegroundPermissionsAsync();
   190→   190→        if (status !== 'granted') {
   191→   191→          setIsLoadingLocation(false);
   192→   192→          return;
   193→   193→        }
   194→   194→
   195→   195→        const loc = await Location.getCurrentPositionAsync({
   196→   196→          accuracy: Location.Accuracy.High,
   197→   197→        });
   198→   198→        setCurrentLocation({
   199→   199→          lat: loc.coords.latitude,
   200→   200→          lng: loc.coords.longitude,
   201→   201→        });
   202→   202→      } catch (e) {
   203→   203→        console.warn('[Map] Failed to get current location:', e);
   204→   204→      } finally {
   205→   205→        setIsLoadingLocation(false);
   206→   206→      }
   207→   207→    })();
   208→   208→  }, []);
   209→   209→
   210→   210→  // Load GPS route points from offline storage
   211→   211→  useEffect(() => {
   212→   212→    (async () => {
   213→   213→      try {
   214→   214→        const locations: GPSPoint[] = await StorageService.getOfflineRouteLocations();
   215→   215→        if (locations.length > 0) {
   216→   216→          const points = locations
   217→   217→            .filter((l) => l.lat && l.lng)
   218→   218→            .map((l) => ({ latitude: l.lat, longitude: l.lng }));
   219→   219→          setRoutePoints(points);
   220→   220→        }
   221→   221→      } catch (e) {
   222→   222→        console.warn('[Map] Failed to load route points:', e);
   223→   223→      }
   224→   224→    })();
   225→   225→  }, [isTracking]);
   226→   226→
   227→   227→  // Load visited shops
   228→   228→  useEffect(() => {
   229→   229→    (async () => {
   230→   230→      const cached = await StorageService.getVisitedShops();
   231→   231→      if (cached.length > 0) setVisitedShopIds(new Set(cached));
   232→   232→    })();
   233→   233→  }, []);
   234→   234→
   235→   235→  // Center map on current location
   236→   236→  const centerOnUser = useCallback(() => {
   237→   237→    if (currentLocation && mapRef.current) {
   238→   238→      mapRef.current.animateToRegion({
   239→   239→        latitude: currentLocation.lat,
   240→   240→        longitude: currentLocation.lng,
   241→   241→        latitudeDelta: 0.01,
   242→   242→        longitudeDelta: 0.01,
   243→   243→      }, 500);
   244→   244→    }
   245→   245→  }, [currentLocation]);
   246→   246→
   247→   247→  // Fit all shops in view
   248→   248→  const fitAllShops = useCallback(() => {
   249→   249→    if (todayShops.length === 0 || !mapRef.current) return;
   250→   250→
   251→   251→    const shopsWithCoords = todayShops.filter((s) => s.lat && s.lng);
   252→   252→    if (shopsWithCoords.length === 0) return;
   253→   253→
   254→   254→    if (shopsWithCoords.length === 1) {
   255→   255→      mapRef.current.animateToRegion({
   256→   256→        latitude: shopsWithCoords[0].lat!,
   257→   257→        longitude: shopsWithCoords[0].lng!,
   258→   258→        latitudeDelta: 0.01,
   259→   259→        longitudeDelta: 0.01,
   260→   260→      }, 500);
   261→   261→      return;
   262→   262→    }
   263→   263→
   264→   264→    // Calculate bounding box
   265→   265→    const latitudes = shopsWithCoords.map((s) => s.lat!);
   266→   266→    const longitudes = shopsWithCoords.map((s) => s.lng!);
   267→   267→    const minLat = Math.min(...latitudes);
   268→   268→    const maxLat = Math.max(...latitudes);
   269→   269→    const minLng = Math.min(...longitudes);
   270→   270→    const maxLng = Math.max(...longitudes);
   271→   271→
   272→   272→    mapRef.current.animateToRegion({
   273→   273→      latitude: (minLat + maxLat) / 2,
   274→   274→      longitude: (minLng + maxLng) / 2,
   275→   275→      latitudeDelta: (maxLat - minLat) * 1.3 + 0.01,
   276→   276→      longitudeDelta: (maxLng - minLng) * 1.3 + 0.01,
   277→   277→    }, 500);
   278→   278→  }, [todayShops]);
   279→   279→
   280→   280→  // Shops with GPS coordinates
   281→   281→  const shopsWithCoords = useMemo(() => {
   282→   282→    const shops = showVisitedOnly
   283→   283→      ? todayShops.filter((s) => visitedShopIds.has(s.id))
   284→   284→      : todayShops;
   285→   285→    return shops.filter((s) => s.lat && s.lng);
   286→   286→  }, [todayShops, showVisitedOnly, visitedShopIds]);
   287→   287→
   288→   288→  // Initial region — always provide a valid default
   289→   289→  const initialRegion = useMemo(() => {
   290→   290→    if (currentLocation) {
   291→   291→      return {
   292→   292→        latitude: currentLocation.lat,
   293→   293→        longitude: currentLocation.lng,
   294→   294→        latitudeDelta: 0.02,
   295→   295→        longitudeDelta: 0.02,
   296→   296→      };
   297→   297→    }
   298→   298→    if (shopsWithCoords.length > 0) {
   299→   299→      const latitudes = shopsWithCoords.map((s) => s.lat!);
   300→   300→      const longitudes = shopsWithCoords.map((s) => s.lng!);
   301→   301→      return {
   302→   302→        latitude: (Math.min(...latitudes) + Math.max(...latitudes)) / 2,
   303→   303→        longitude: (Math.min(...longitudes) + Math.max(...longitudes)) / 2,
   304→   304→        latitudeDelta: 0.05,
   305→   305→        longitudeDelta: 0.05,
   306→   306→      };
   307→   307→    }
   308→   308→    return {
   309→   309→      latitude: DEFAULT_LAT,
   310→   310→      longitude: DEFAULT_LNG,
   311→   311→      latitudeDelta: 0.5,
   312→   312→      longitudeDelta: 0.5,
   313→   313→    };
   314→   314→  }, [currentLocation, shopsWithCoords]);
   315→   315→
   316→   316→  // Get distance from current location to shop
   317→   317→  const getDistanceToShop = useCallback((shop: Shop): string | null => {
   318→   318→    if (!currentLocation || !shop.lat || !shop.lng) return null;
   319→   319→    const meters = getDistanceMeters(currentLocation.lat, currentLocation.lng, shop.lat, shop.lng);
   320→   320→    if (meters < 1000) return `${Math.round(meters)}m`;
   321→   321→    return `${(meters / 1000).toFixed(1)}km`;
   322→   322→  }, [currentLocation]);
   323→   323→
   324→   324→  // Get shop balance for display
   325→   325→  const getShopBalance = useCallback((shop: Shop): number => {
   326→   326→    try {
   327→   327→      if (shop.companyBalances && shop.companyBalances.length > 0 && selectedCompanyId) {
   328→   328→        const companyBal = shop.companyBalances.find((cb) => cb.companyId === selectedCompanyId);
   329→   329→        if (companyBal) return companyBal.balance;
   330→   330→      }
   331→   331→      return shop.balance || 0;
   332→   332→    } catch {
   333→   333→      return shop.balance || 0;
   334→   334→    }
   335→   335→  }, [selectedCompanyId]);
   336→   336→
   337→   337→  // Stats
   338→   338→  const totalShops = todayShops.length;
   339→   339→  const shopsOnMap = shopsWithCoords.length;
   340→   340→  const visitedCount = visitedShopIds.size;
   341→   341→
   342→   342→  // Ensure MapViewComponent is available
   343→   343→  if (!MapViewComponent) {
   344→   344→    return <MapFallback />;
   345→   345→  }
   346→   346→
   347→   347→  return (
   348→   348→    <View style={[styles.root, { paddingTop: insets.top }]}>
   349→   349→      {/* Map Header */}
   350→   350→      <View style={styles.header}>
   351→   351→        <View style={styles.headerLeft}>
   352→   352→          <MaterialIcons name="map" size={22} color={Colors.primary} />
   353→   353→          <Text style={styles.headerTitle}>Route Map</Text>
   354→   354→        </View>

   375→   375→      {/* Route Status Bar */}
   376→   376→      {isTracking && (
   377→   377→        <View style={styles.routeStatusBar}>
   378→   378→          <View style={styles.routeStatusDot} />
   379→   379→          <Text style={styles.routeStatusText}>
   380→   380→            Route Active — {routePoints.length} GPS points
   381→   381→          </Text>
   382→   382→          <Text style={styles.routeStatusTime}>
   383→   383→            Since {startTime ? new Date(startTime).toLocaleTimeString('en-PK', { hour: '2-digit', minute: '2-digit', hour12: true, timeZone: 'Asia/Karachi' }) : '--:--'}

   396→   396→        showsMyLocationButton={false}
   397→   397→        showsCompass={true}
   398→   398→        showsScale={true}
   399→   399→        loadingEnabled={true}
   400→   400→        loadingBackgroundColor={Colors.background}
   401→   401→        loadingIndicatorColor={Colors.primary}

   406→   406→          const distance = getDistanceToShop(shop);
   407→   407→          const balance = getShopBalance(shop);
   408→   408→

   411→   411→              {/* 30m Proximity Circle */}

   448→   448→        {currentLocation && !isTracking && CircleComponent && (

   450→   450→            center={{ latitude: currentLocation.lat, longitude: currentLocation.lng }}

   546→   546→        </View>
