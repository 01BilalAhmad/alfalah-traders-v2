    44→// Error Boundary class component — catches React render errors that try-catch cannot
    45→class MapErrorBoundary extends Component<
    46→  { children: React.ReactNode; onError: () => void },
    47→  { hasError: boolean; error: Error | null }
    48→> {
    49→  state = { hasError: false, error: null as Error | null };
    50→
    51→  static getDerivedStateFromError(error: Error) {
    52→    return { hasError: true, error };
    53→  }
    54→
    55→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    56→    console.error('[Map] ErrorBoundary caught:', error, info);
    57→    // Notify parent so it can show fallback
    58→    this.props.onError();
    59→  }
    60→
    61→  render() {
    62→    if (this.state.hasError) {
    63→      return null; // Parent will show fallback
    64→    }
    65→    return this.props.children;
    66→  }
    67→}

    69→// Lazy load MapView and related components to prevent native crash from killing the whole app
    70→let MapViewComponent: any = null;

    72→let PolylineComponent: any = null;

    76→async function loadMapComponents() {
    77→  try {
    78→    const maps = await import('react-native-maps');
    79→    MapViewComponent = maps.default;
    80→    MarkerComponent = maps.Marker;
    81→    PolylineComponent = maps.Polyline;
    82→    CircleComponent = maps.Circle;
    83→    PROVIDER_DEFAULT_CONST = maps.PROVIDER_DEFAULT;
    84→    return true;
    85→  } catch (e) {
    86→    console.error('[Map] Failed to load react-native-maps:', e);
    87→    return false;
    88→  }
    89→}
    90→
    91→export default function MapScreen() {

    93→  const [mapFailed, setMapFailed] = useState(false);
    94→
    95→  useEffect(() => {
    96→    loadMapComponents().then((success) => {
    97→      if (success) {
    98→        setMapLoaded(true);
    99→      } else {
   100→        setMapFailed(true);
   101→      }
   102→    }).catch(() => {
   103→      setMapFailed(true);
   104→    });
   105→  }, []);
   106→
   107→  if (mapFailed) {
   108→    return <MapFallback />;
   109→  }

   122→  return (
   123→    <MapErrorBoundary onError={() => setMapFailed(true)}>
   124→      <MapScreenInner />
   125→    </MapErrorBoundary>
   126→  );
   127→}

   169→function MapScreenInner() {
   170→  const insets = useSafeAreaInsets();
   171→  const { user, selectedCompanyId } = useAuth();
   172→  const { todayShops, isOnline } = useShops();
   173→  const { isTracking, startTime } = useRouteTracking();
   174→
   175→  const mapRef = useRef<any>(null);
   176→  const [currentLocation, setCurrentLocation] = useState<{ lat: number; lng: number } | null>(null);
   177→  const [routePoints, setRoutePoints] = useState<Array<{ latitude: number; longitude: number }>>([]);
   178→  const [selectedShop, setSelectedShop] = useState<Shop | null>(null);
   179→  const [isLoadingLocation, setIsLoadingLocation] = useState(true);
   180→  const [mapType, setMapType] = useState<'standard' | 'satellite'>('standard');
   181→  const [showVisitedOnly, setShowVisitedOnly] = useState(false);
   182→  const [visitedShopIds, setVisitedShopIds] = useState<Set<string>>(new Set());
   183→
   184→  // Load current location
   185→  useEffect(() => {
   186→    (async () => {
   187→      try {
   188→        const Location = require('expo-location');
   189→        const { status } = await Location.requestForegroundPermissionsAsync();
   190→        if (status !== 'granted') {
   191→          setIsLoadingLocation(false);
   192→          return;
   193→        }
   194→
   195→        const loc = await Location.getCurrentPositionAsync({
   196→          accuracy: Location.Accuracy.High,
   197→        });
   198→        setCurrentLocation({
   199→          lat: loc.coords.latitude,
   200→          lng: loc.coords.longitude,
   201→        });
   202→      } catch (e) {
   203→        console.warn('[Map] Failed to get current location:', e);
   204→      } finally {
   205→        setIsLoadingLocation(false);
   206→      }
   207→    })();
   208→  }, []);
   209→
   210→  // Load GPS route points from offline storage
   211→  useEffect(() => {
   212→    (async () => {
   213→      try {
   214→        const locations: GPSPoint[] = await StorageService.getOfflineRouteLocations();
   215→        if (locations.length > 0) {
   216→          const points = locations
   217→            .filter((l) => l.lat && l.lng)
   218→            .map((l) => ({ latitude: l.lat, longitude: l.lng }));
   219→          setRoutePoints(points);
   220→        }
   221→      } catch (e) {
   222→        console.warn('[Map] Failed to load route points:', e);
   223→      }
   224→    })();
   225→  }, [isTracking]);
   226→
   227→  // Load visited shops
   228→  useEffect(() => {
   229→    (async () => {
   230→      const cached = await StorageService.getVisitedShops();
   231→      if (cached.length > 0) setVisitedShopIds(new Set(cached));
   232→    })();
   233→  }, []);
   234→
   235→  // Center map on current location
   236→  const centerOnUser = useCallback(() => {
   237→    if (currentLocation && mapRef.current) {
   238→      mapRef.current.animateToRegion({
   239→        latitude: currentLocation.lat,
   240→        longitude: currentLocation.lng,
   241→        latitudeDelta: 0.01,
   242→        longitudeDelta: 0.01,
   243→      }, 500);
   244→    }
   245→  }, [currentLocation]);
   246→
   247→  // Fit all shops in view
   248→  const fitAllShops = useCallback(() => {
   249→    if (todayShops.length === 0 || !mapRef.current) return;
   250→
   251→    const shopsWithCoords = todayShops.filter((s) => s.lat && s.lng);
   252→    if (shopsWithCoords.length === 0) return;
   253→
   254→    if (shopsWithCoords.length === 1) {
   255→      mapRef.current.animateToRegion({
   256→        latitude: shopsWithCoords[0].lat!,
   257→        longitude: shopsWithCoords[0].lng!,
   258→        latitudeDelta: 0.01,
   259→        longitudeDelta: 0.01,
   260→      }, 500);
   261→      return;
   262→    }
   263→
   264→    // Calculate bounding box
   265→    const latitudes = shopsWithCoords.map((s) => s.lat!);
   266→    const longitudes = shopsWithCoords.map((s) => s.lng!);
   267→    const minLat = Math.min(...latitudes);
   268→    const maxLat = Math.max(...latitudes);
   269→    const minLng = Math.min(...longitudes);
   270→    const maxLng = Math.max(...longitudes);
   271→
   272→    mapRef.current.animateToRegion({
   273→      latitude: (minLat + maxLat) / 2,
   274→      longitude: (minLng + maxLng) / 2,
   275→      latitudeDelta: (maxLat - minLat) * 1.3 + 0.01,
   276→      longitudeDelta: (maxLng - minLng) * 1.3 + 0.01,
   277→    }, 500);
   278→  }, [todayShops]);
   279→
   280→  // Shops with GPS coordinates
   281→  const shopsWithCoords = useMemo(() => {
   282→    const shops = showVisitedOnly
   283→      ? todayShops.filter((s) => visitedShopIds.has(s.id))
   284→      : todayShops;
   285→    return shops.filter((s) => s.lat && s.lng);
   286→  }, [todayShops, showVisitedOnly, visitedShopIds]);
   287→
   288→  // Initial region — always provide a valid default
   289→  const initialRegion = useMemo(() => {
   290→    if (currentLocation) {
   291→      return {
   292→        latitude: currentLocation.lat,
   293→        longitude: currentLocation.lng,
   294→        latitudeDelta: 0.02,
   295→        longitudeDelta: 0.02,
   296→      };
   297→    }
   298→    if (shopsWithCoords.length > 0) {
   299→      const latitudes = shopsWithCoords.map((s) => s.lat!);
   300→      const longitudes = shopsWithCoords.map((s) => s.lng!);
   301→      return {
   302→        latitude: (Math.min(...latitudes) + Math.max(...latitudes)) / 2,
   303→        longitude: (Math.min(...longitudes) + Math.max(...longitudes)) / 2,
   304→        latitudeDelta: 0.05,
   305→        longitudeDelta: 0.05,
   306→      };
   307→    }
   308→    return {
   309→      latitude: DEFAULT_LAT,
   310→      longitude: DEFAULT_LNG,
   311→      latitudeDelta: 0.5,
   312→      longitudeDelta: 0.5,
   313→    };
   314→  }, [currentLocation, shopsWithCoords]);
   315→
   316→  // Get distance from current location to shop
   317→  const getDistanceToShop = useCallback((shop: Shop): string | null => {
   318→    if (!currentLocation || !shop.lat || !shop.lng) return null;
   319→    const meters = getDistanceMeters(currentLocation.lat, currentLocation.lng, shop.lat, shop.lng);
   320→    if (meters < 1000) return `${Math.round(meters)}m`;
   321→    return `${(meters / 1000).toFixed(1)}km`;
   322→  }, [currentLocation]);
   323→
   324→  // Get shop balance for display
   325→  const getShopBalance = useCallback((shop: Shop): number => {
   326→    try {
   327→      if (shop.companyBalances && shop.companyBalances.length > 0 && selectedCompanyId) {
   328→        const companyBal = shop.companyBalances.find((cb) => cb.companyId === selectedCompanyId);
   329→        if (companyBal) return companyBal.balance;
   330→      }
   331→      return shop.balance || 0;
   332→    } catch {
   333→      return shop.balance || 0;
   334→    }
   335→  }, [selectedCompanyId]);
   336→
   337→  // Stats
   338→  const totalShops = todayShops.length;
   339→  const shopsOnMap = shopsWithCoords.length;
   340→  const visitedCount = visitedShopIds.size;
   341→
   342→  // Ensure MapViewComponent is available
   343→  if (!MapViewComponent) {
   344→    return <MapFallback />;
   345→  }
   346→
   347→  return (
   348→    <View style={[styles.root, { paddingTop: insets.top }]}>
   349→      {/* Map Header */}
   350→      <View style={styles.header}>
   351→        <View style={styles.headerLeft}>
   352→          <MaterialIcons name="map" size={22} color={Colors.primary} />
   353→          <Text style={styles.headerTitle}>Route Map</Text>
   354→        </View>
   355→        <View style={styles.headerRight}>
   356→          <Pressable
   357→            style={[styles.headerBtn, showVisitedOnly && styles.headerBtnActive]}
   358→            onPress={() => setShowVisitedOnly((v) => !v)}
   359→          >
   360→            <MaterialIcons
   361→              name={showVisitedOnly ? 'visibility' : 'visibility-off'}
   362→              size={18}
   363→              color={showVisitedOnly ? Colors.primary : Colors.textSecondary}
   364→            />
   365→          </Pressable>
   366→          <Pressable
   367→            style={styles.headerBtn}
   368→            onPress={() => setMapType((t) => t === 'standard' ? 'satellite' : 'standard')}
   369→          >
   370→            <MaterialIcons name="layers" size={18} color={Colors.textSecondary} />
   371→          </Pressable>
   372→        </View>
   373→      </View>
   374→
   375→      {/* Route Status Bar */}
   376→      {isTracking && (
   377→        <View style={styles.routeStatusBar}>
   378→          <View style={styles.routeStatusDot} />
   379→          <Text style={styles.routeStatusText}>
   380→            Route Active — {routePoints.length} GPS points
   381→          </Text>
   382→          <Text style={styles.routeStatusTime}>
   383→            Since {startTime ? new Date(startTime).toLocaleTimeString('en-PK', { hour: '2-digit', minute: '2-digit', hour12: true, timeZone: 'Asia/Karachi' }) : '--:--'}
   384→          </Text>
   385→        </View>
   386→      )}
   387→
   388→      {/* Map View — dynamically loaded */}
   389→      <MapViewComponent
   390→        ref={mapRef}
   391→        style={styles.map}
   392→        provider={PROVIDER_DEFAULT_CONST}
   393→        initialRegion={initialRegion}
   394→        mapType={mapType}
   395→        showsUserLocation={true}
   396→        showsMyLocationButton={false}
   397→        showsCompass={true}
   398→        showsScale={true}
   399→        loadingEnabled={true}
   400→        loadingBackgroundColor={Colors.background}
   401→        loadingIndicatorColor={Colors.primary}
   402→      >
   403→        {/* Shop Markers */}
   404→        {shopsWithCoords.map((shop) => {
   405→          const isVisited = visitedShopIds.has(shop.id);
   406→          const distance = getDistanceToShop(shop);
   407→          const balance = getShopBalance(shop);
   408→
   409→          return (
   410→            <React.Fragment key={shop.id}>
   411→              {/* 30m Proximity Circle */}
   412→              {CircleComponent && (
   413→                <CircleComponent
   414→                  center={{ latitude: shop.lat!, longitude: shop.lng! }}
   415→                  radius={PROXIMITY_RADIUS}
   416→                  strokeColor={isVisited ? '#10B981' : '#6366F180'}
   417→                  fillColor={isVisited ? '#10B98115' : '#6366F108'}
   418→                  strokeWidth={1}
   419→                />
   420→              )}
   421→              {/* Shop Marker */}
   422→              {MarkerComponent && (
   423→                <MarkerComponent
   424→                  coordinate={{ latitude: shop.lat!, longitude: shop.lng! }}
   425→                  onPress={() => setSelectedShop(shop)}
   426→                  pinColor={isVisited ? '#10B981' : '#6366F1'}
   427→                  title={shop.name}
   428→                  description={`${shop.area} • ${formatPKR(balance)}`}
   429→                />
   430→              )}
   431→            </React.Fragment>
   432→          );
   433→        })}
   434→
   435→        {/* Route Polyline (GPS waypoints) */}
   436→        {routePoints.length > 1 && PolylineComponent && (
   437→          <PolylineComponent
   438→            coordinates={routePoints}
   439→            strokeColor="#4F46E5"
   440→            strokeWidth={4}
   441→            lineCap="round"
   442→            lineJoin="round"
   443→            fillColor="#4F46E520"
   444→          />
   445→        )}
   446→
   447→        {/* Current Location Marker (custom blue dot) — shown if no native blue dot */}
   448→        {currentLocation && !isTracking && CircleComponent && (
   449→          <CircleComponent
   450→            center={{ latitude: currentLocation.lat, longitude: currentLocation.lng }}
   451→            radius={5}
   452→            strokeColor="#3B82F6"
   453→            fillColor="#3B82F6"
   454→          />
   455→        )}
   456→      </MapViewComponent>
   457→
   458→      {/* Floating Action Buttons */}
   459→      <View style={[styles.fabContainer, { bottom: insets.bottom + 90 }]}>
   460→        <Pressable style={styles.fabBtn} onPress={centerOnUser}>
   461→          <MaterialIcons name="my-location" size={22} color={Colors.primary} />
   462→        </Pressable>
   463→        <Pressable style={styles.fabBtn} onPress={fitAllShops}>
   464→          <MaterialIcons name="zoom-out-map" size={22} color={Colors.primary} />
   465→        </Pressable>
   466→      </View>
   467→
   468→      {/* Map Stats Overlay */}
   469→      <View style={[styles.statsOverlay, { bottom: insets.bottom + 90 }]}>
   470→        <View style={styles.statsRow}>
   471→          <View style={styles.statChip}>
   472→            <MaterialIcons name="store" size={14} color={Colors.primary} />
   473→            <Text style={styles.statText}>{shopsOnMap}/{totalShops}</Text>
   474→          </View>
   475→          <View style={[styles.statChip, { backgroundColor: '#ECFDF5' }]}>
   476→            <MaterialIcons name="check-circle" size={14} color="#10B981" />
   477→            <Text style={[styles.statText, { color: '#065F46' }]}>{visitedCount} visited</Text>
   478→          </View>
   479→          {routePoints.length > 0 && (
   480→            <View style={[styles.statChip, { backgroundColor: '#EEF2FF' }]}>
   481→              <MaterialIcons name="route" size={14} color="#4F46E5" />
   482→              <Text style={[styles.statText, { color: '#3730A3' }]}>{routePoints.length} pts</Text>
   483→            </View>
   484→          )}
   485→        </View>
   486→      </View>
   487→
   488→      {/* Selected Shop Info Card */}
   489→      {selectedShop && (
   490→        <View style={[styles.shopCard, { bottom: insets.bottom + 140 }]}>
   491→          <Pressable style={styles.shopCardClose} onPress={() => setSelectedShop(null)}>
   492→            <MaterialIcons name="close" size={18} color={Colors.textSecondary} />
   493→          </Pressable>
   494→          <View style={styles.shopCardHeader}>
   495→            <View style={[styles.shopPin, { backgroundColor: visitedShopIds.has(selectedShop.id) ? '#10B981' : '#6366F1' }]}>
   496→              <MaterialIcons name="store" size={16} color="#FFFFFF" />
   497→            </View>
   498→            <View style={styles.shopCardInfo}>
   499→              <Text style={styles.shopCardName} numberOfLines={1}>{selectedShop.name}</Text>
   500→              <Text style={styles.shopCardArea} numberOfLines={1}>{selectedShop.area}</Text>
   501→            </View>
   502→          </View>
   503→          <View style={styles.shopCardDetails}>
   504→            <View style={styles.shopDetailRow}>
   505→              <MaterialIcons name="account-balance-wallet" size={16} color={Colors.primary} />
   506→              <Text style={styles.shopDetailText}>Balance: {formatPKR(getShopBalance(selectedShop))}</Text>
   507→            </View>
   508→            {selectedShop.ownerName ? (
   509→              <View style={styles.shopDetailRow}>
   510→                <MaterialIcons name="person" size={16} color={Colors.textSecondary} />
   511→                <Text style={styles.shopDetailText}>{selectedShop.ownerName}</Text>
   512→              </View>
   513→            ) : null}
   514→            {getDistanceToShop(selectedShop) && (
   515→              <View style={styles.shopDetailRow}>
   516→                <MaterialIcons name="straighten" size={16} color={Colors.textSecondary} />
   517→                <Text style={styles.shopDetailText}>{getDistanceToShop(selectedShop)} away</Text>
   518→              </View>
   519→            )}
   520→            {selectedShop.phone ? (
   521→              <View style={styles.shopDetailRow}>
   522→                <MaterialIcons name="phone" size={16} color={Colors.textSecondary} />
   523→                <Text style={styles.shopDetailText}>{selectedShop.phone}</Text>
   524→              </View>
   525→            ) : null}
   526→          </View>
   527→          <View style={styles.shopCardBadge}>
   528→            <View style={[styles.badgeDot, { backgroundColor: visitedShopIds.has(selectedShop.id) ? '#10B981' : '#9CA3AF' }]} />
   529→            <Text style={[styles.badgeText, { color: visitedShopIds.has(selectedShop.id) ? '#065F46' : '#6B7280' }]}>
   530→              {visitedShopIds.has(selectedShop.id) ? 'Visited' : 'Not Visited'}
   531→            </Text>
   532→          </View>
   533→        </View>
   534→      )}
   535→
   536→      {/* No Shops Warning */}
   537→      {shopsOnMap === 0 && !isLoadingLocation && (
   538→        <View style={styles.noDataOverlay}>
   539→          <MaterialIcons name="location-off" size={48} color={Colors.textMuted} />
   540→          <Text style={styles.noDataTitle}>No Shop Locations</Text>
   541→          <Text style={styles.noDataDesc}>
   542→            {totalShops === 0
   543→              ? 'No shops loaded. Download data first.'
   544→              : `${totalShops} shops found but none have GPS coordinates set.`}
   545→          </Text>
   546→        </View>
   547→      )}
   548→    </View>
   549→  );
   550→}

   552→const styles = StyleSheet.create({
