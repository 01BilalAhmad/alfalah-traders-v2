   198→  // --- Offline Data Expiry (7 Day Auto-Clear) ---
   199→  cleanExpiredOfflineQueue: async (): Promise<number> => {
   200→    const raw = await AsyncStorage.getItem(KEYS.OFFLINE_QUEUE);
   201→    const queue: OfflineRecovery[] = raw ? JSON.parse(raw) : [];
   202→    const SEVEN_DAYS_MS = 7 * 24 * 60 * 60 * 1000;
   203→    const cutoff = Date.now() - SEVEN_DAYS_MS;
   204→    const before = queue.length;
   205→    const filtered = queue.filter((r) => {
   206→      const created = new Date(r.createdAt).getTime();
   207→      return created >= cutoff;
   208→    });
   209→    const removed = before - filtered.length;
   210→    if (removed > 0) {
   211→      await AsyncStorage.setItem(KEYS.OFFLINE_QUEUE, JSON.stringify(filtered));
   212→    }
   213→    return removed;
   214→  },

   337→  updateVisitStreak: async (orderbookerId: string, visitedToday: boolean): Promise<VisitStreak> => {
   338→    const today = new Date().toISOString().split('T')[0]; // YYYY-MM-DD
   339→    const existing = await StorageService.getVisitStreak(orderbookerId);
   340→
   341→    if (!existing) {
   342→      const newStreak: VisitStreak = {
   343→        orderbookerId,
   344→        currentStreak: visitedToday ? 1 : 0,
   345→        lastVisitDate: visitedToday ? today : '',
   346→        longestStreak: visitedToday ? 1 : 0,
   347→      };
   348→      await StorageService.saveVisitStreak(newStreak);
   349→      return newStreak;
   350→    }
   351→
   352→    // Already updated today
   353→    if (existing.lastVisitDate === today) {
   354→      return existing;
   355→    }
   356→
   357→    if (visitedToday) {
   358→      // Check if last visit was yesterday (consecutive)
   359→      const lastVisit = new Date(existing.lastVisitDate);
   360→      const todayDate = new Date(today);
   361→      const diffDays = Math.round((todayDate.getTime() - lastVisit.getTime()) / (1000 * 60 * 60 * 24));
   362→
   363→      const newCurrentStreak = diffDays <= 1 ? existing.currentStreak + 1 : 1;
   364→      const newLongestStreak = Math.max(existing.longestStreak, newCurrentStreak);
   365→
   366→      const updated: VisitStreak = {
   367→        ...existing,
   368→        currentStreak: newCurrentStreak,
   369→        lastVisitDate: today,
   370→        longestStreak: newLongestStreak,
   371→      };
   372→      await StorageService.saveVisitStreak(updated);
   373→      return updated;
   374→    } else {
   375→      // Not visited today — check if streak should be broken
   376→      const lastVisit = new Date(existing.lastVisitDate);
   377→      const todayDate = new Date(today);
   378→      const diffDays = Math.round((todayDate.getTime() - lastVisit.getTime()) / (1000 * 60 * 60 * 24));
   379→
   380→      if (diffDays > 1) {
   381→        // Streak broken — reset
   382→        const updated: VisitStreak = {
   383→          ...existing,
   384→          currentStreak: 0,
   385→        };
   386→        await StorageService.saveVisitStreak(updated);
   387→        return updated;
   388→      }
   389→      return existing;
   390→    }
   391→  },

   426→  incrementNotifCount: async (method: 'sms' | 'whatsapp', shopId?: string): Promise<{ sms: number; whatsapp: number }> => {
   427→    try {
   428→      const raw = await AsyncStorage.getItem(KEYS.NOTIF_COUNTS);
   429→      const shopsRaw = await AsyncStorage.getItem(KEYS.NOTIF_SHOPS);
   430→      let entry: { date: string; sms: number; whatsapp: number };
   431→      let shopsEntry: { date: string; smsShops: string[]; whatsappShops: string[] };
   432→
   433→      if (!raw) {
   434→        entry = { date: getTodayDateStr(), sms: 0, whatsapp: 0 };
   435→      } else {
   436→        entry = JSON.parse(raw);
   437→        if (entry.date !== getTodayDateStr()) {
   438→          entry = { date: getTodayDateStr(), sms: 0, whatsapp: 0 };
   439→        }
   440→      }
   441→
   442→      if (!shopsRaw) {
   443→        shopsEntry = { date: getTodayDateStr(), smsShops: [], whatsappShops: [] };
   444→      } else {
   445→        shopsEntry = JSON.parse(shopsRaw);
   446→        if (shopsEntry.date !== getTodayDateStr()) {
   447→          shopsEntry = { date: getTodayDateStr(), smsShops: [], whatsappShops: [] };
   448→        }
   449→      }
   450→
   451→      // Only increment count if this shop hasn't been counted before for this method
   452→      if (shopId) {
   453→        const shopKey = method === 'sms' ? 'smsShops' : 'whatsappShops';
   454→        if (!shopsEntry[shopKey].includes(shopId)) {
   455→          shopsEntry[shopKey].push(shopId);
   456→          entry[method] = (entry[method] || 0) + 1;
   457→        }
   458→      } else {
   459→        // Fallback: no shopId, just increment
   460→        entry[method] = (entry[method] || 0) + 1;
   461→      }
   462→
   463→      await AsyncStorage.setItem(KEYS.NOTIF_COUNTS, JSON.stringify(entry));
   464→      await AsyncStorage.setItem(KEYS.NOTIF_SHOPS, JSON.stringify(shopsEntry));
   465→      return { sms: entry.sms, whatsapp: entry.whatsapp };
   466→    } catch {
   467→      return { sms: 0, whatsapp: 0 };
   468→    }
   469→  },

   784→  // --- Check if yesterday's data was synced ---
   785→  // Returns true if yesterday's sync upload is done OR if there's no pending data
   786→  isYesterdaySyncDone: async (): Promise<boolean> => {
   787→    const yesterday = new Date();
   788→    yesterday.setDate(yesterday.getDate() - 1);
   789→    const yesterdayStr = yesterday.toISOString().split('T')[0]; // YYYY-MM-DD
   790→
   791→    const lastUpload = await AsyncStorage.getItem(KEYS.LAST_SYNC_UPLOAD_DATE);
   792→    const downloadedFor = await AsyncStorage.getItem(KEYS.DATA_DOWNLOADED_FOR_DATE);
   793→
   794→    // If we never downloaded data for yesterday, no sync needed
   795→    if (downloadedFor !== yesterdayStr) return true;
   796→
   797→    // If we downloaded for yesterday, check if upload was done
   798→    if (lastUpload && lastUpload >= yesterdayStr) return true;
   799→
   800→    return false;
   801→  },
