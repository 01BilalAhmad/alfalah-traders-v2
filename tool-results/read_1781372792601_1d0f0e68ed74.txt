   115→// ── Time/Duration Formatting ───────────────────────────────────────────────
   116→function formatTime(isoString: string): string {
   117→  try {
   118→    return new Date(isoString).toLocaleTimeString('en-PK', {
   119→      timeZone: 'Asia/Karachi',
   120→      hour: '2-digit',
   121→      minute: '2-digit',
   122→      hour12: true,
   123→    });
   124→  } catch {
   125→    return '--:--';
   126→  }
   127→}
   128→
   129→function formatDuration(seconds: number | null): string {
   130→  if (!seconds) return '--';
   131→  const h = Math.floor(seconds / 3600);
   132→  const m = Math.floor((seconds % 3600) / 60);
   133→  if (h > 0) return `${h}h ${m}m`;
   134→  return `${m}m`;
   135→}
   136→
   137→function formatMinutes(seconds: number | null): string {
   138→  if (!seconds) return '--';
   139→  const m = Math.round(seconds / 60);
   140→  if (m < 1) return '<1m';
   141→  return `${m}m`;
   142→}
   143→
   144→function calcDuration(startTime: string, endTime?: string | null): number {
   145→  const start = new Date(startTime).getTime();
   146→  const end = endTime ? new Date(endTime).getTime() : Date.now();
   147→  return Math.max(0, Math.round((end - start) / 1000));
   148→}
