     2→import React, { useEffect, useState, memo } from 'react';
     3→import {
     4→  View,
     5→  Text,
     6→  StyleSheet,
     7→  Modal,
     8→  Pressable,
     9→  ScrollView,
    10→  ActivityIndicator,
    11→  Linking,
    12→  Dimensions,
    13→  TextInput,
    14→  Alert,
    15→  KeyboardAvoidingView,
    16→  Platform,
    17→} from 'react-native';
    18→import { MaterialIcons } from '@expo/vector-icons';
    19→import { LinearGradient } from 'expo-linear-gradient';
    20→import { BarChart } from 'react-native-chart-kit';
    21→import { Colors, Spacing, Radius, FontSize, FontWeight, Shadow } from '@/constants/theme';
    22→import { Shop, Transaction, ApiService } from '@/services/api';
    23→import { getShopDisplayBalance } from '@/components/ui/ShopCard';
    24→import { formatPKR, formatDateTime } from '@/utils/format';
    25→import { CreditBar } from './CreditBar';
    26→import { Badge } from './Badge';
    27→import { StorageService, ShopNote } from '@/services/storage';
    28→
    29→const screenWidth = Dimensions.get('window').width;
    30→
    31→interface ShopDetailModalProps {
    32→  visible: boolean;
    33→  shop: Shop | null;
    34→  companyId?: string;
    35→  onClose: () => void;
    36→  onCollect: () => void;
    37→  hasRecoveryToday?: boolean; // Whether recovery was submitted for this shop today
    38→  onResendReceipt?: () => void; // Callback to resend receipt
    39→  onEditPendingRecovery?: (txn: Transaction) => void; // Callback to edit a pending recovery
    40→}
    41→
    42→export const ShopDetailModal = memo(function ShopDetailModal({
    43→  visible,
    44→  shop,
    45→  companyId,
    46→  onClose,
    47→  onCollect,
    48→  hasRecoveryToday,
    49→  onResendReceipt,
    50→  onEditPendingRecovery,
    51→}: ShopDetailModalProps) {
    52→  const [recentTxns, setRecentTxns] = useState<Transaction[]>([]);
    53→  const [loading, setLoading] = useState(false);
    54→  const [chartData, setChartData] = useState<{ labels: string[]; credits: number[]; recoveries: number[] }>({
    55→    labels: [],
    56→    credits: [],
    57→    recoveries: [],
    58→  });
    59→  const [chartLoading, setChartLoading] = useState(false);
    60→  const [shopNote, setShopNote] = useState<ShopNote | null>(null);
    61→  const [noteInput, setNoteInput] = useState('');
    62→  const [noteSaving, setNoteSaving] = useState(false);
    63→
    64→  // Phone edit state
    65→  const [editingPhone, setEditingPhone] = useState(false);
    66→  const [phoneInput, setPhoneInput] = useState('');
    67→  const [phoneSaving, setPhoneSaving] = useState(false);
    68→  const [localPhone, setLocalPhone] = useState<string | null>(null);
    69→  const [ownerNameInput, setOwnerNameInput] = useState('');
    70→  const [localOwnerName, setLocalOwnerName] = useState<string | null>(null);
    71→
    72→  useEffect(() => {
    73→    if (visible && shop) {
    74→      loadRecent();
    75→      loadChartData();
    76→      loadShopNote();
    77→      setLocalPhone(null);
    78→      setEditingPhone(false);
    79→      setPhoneInput('');
    80→      setOwnerNameInput('');
    81→      setLocalOwnerName(null);
    82→    }
    83→  }, [visible, shop]);
    84→
    85→  async function loadRecent() {
    86→    if (!shop) return;
    87→    setLoading(true);
    88→    try {
    89→      const res = await ApiService.getTransactions({ shopId: shop.id, limit: 8 });
    90→      setRecentTxns(res.transactions);
    91→    } catch {
    92→      setRecentTxns([]);
    93→    } finally {
    94→      setLoading(false);
    95→    }
    96→  }
    97→
    98→  async function loadChartData() {
    99→    if (!shop) return;
   100→    setChartLoading(true);
   101→    try {
   102→      const now = new Date();
   103→      const labels: string[] = [];
   104→      const credits: number[] = [];
   105→      const recoveries: number[] = [];
   106→
   107→      for (let i = 5; i >= 0; i--) {
   108→        const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   109→        const label = d.toLocaleDateString('en-PK', { month: 'short' });
   110→        const startDate = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-01`;
   111→        const endD = new Date(d.getFullYear(), d.getMonth() + 1, 0);
   112→        labels.push(label);
   113→        try {
   114→          const res = await ApiService.getTransactions({ shopId: shop.id, date: startDate, limit: 500 });
   115→          const endDateStr = `${endD.getFullYear()}-${String(endD.getMonth() + 1).padStart(2, '0')}-${String(endD.getDate()).padStart(2, '0')}`;
   116→          const monthTxns = res.transactions.filter((t) => {
   117→            const tDate = t.createdAt.split('T')[0];
   118→            return tDate >= startDate && tDate <= endDateStr;
   119→          });
   120→          const c = monthTxns.filter((t) => t.type === 'credit').reduce((s, t) => s + t.amount, 0);
   121→          const r = monthTxns.filter((t) => t.type === 'recovery').reduce((s, t) => s + t.amount, 0);
   122→          credits.push(c);
   123→          recoveries.push(r);
   124→        } catch {
   125→          credits.push(0);
   126→          recoveries.push(0);
   127→        }
   128→      }
   129→      setChartData({ labels, credits, recoveries });
   130→    } catch {
   131→      setChartData({ labels: [], credits: [], recoveries: [] });
   132→    } finally {
   133→      setChartLoading(false);
   134→    }
   135→  }
   136→
   137→  async function loadShopNote() {
   138→    if (!shop) return;
   139→    try {
   140→      const note = await StorageService.getShopNote(shop.id);
   141→      setShopNote(note);
   142→      setNoteInput(note ? note.note : '');
   143→    } catch {
   144→      setShopNote(null);
   145→      setNoteInput('');
   146→    }
   147→  }
   148→
   149→  async function handleSaveNote() {
   150→    if (!shop) return;
   151→    setNoteSaving(true);
   152→    try {
   153→      await StorageService.saveShopNote(shop.id, noteInput.trim());
   154→      await loadShopNote();
   155→    } catch {
   156→      Alert.alert('Error', 'Could not save note.');
   157→    } finally {
   158→      setNoteSaving(false);
   159→    }
   160→  }
   161→
   162→  async function handleDeleteNote() {
   163→    if (!shop) return;
   164→    Alert.alert('Delete Note', 'Are you sure you want to delete this note?', [
   165→      { text: 'Cancel', style: 'cancel' },
   166→      {
   167→        text: 'Delete',
   168→        style: 'destructive',
   169→        onPress: async () => {
   170→          try {
   171→            await StorageService.deleteShopNote(shop.id);
   172→            setShopNote(null);
   173→            setNoteInput('');
   174→          } catch { /* not critical */ }
   175→        },
   176→      },
   177→    ]);
   178→  }
   179→
   180→  // Current phone: local override or shop prop
   181→  const currentPhone = localPhone ?? shop?.phone ?? '';
   182→  const currentOwnerName = localOwnerName ?? shop?.ownerName ?? '';
   183→
   184→  function handleSmsPress() {
   185→    if (!shop) return;
   186→    if (!currentPhone) {
   187→      Alert.alert(
   188→        'No Phone Number',
   189→        "This shop doesn't have a phone number saved.",
   190→        [
   191→          { text: 'Cancel', style: 'cancel' },
   192→          {
   193→            text: 'Add Phone Number',
   194→            onPress: () => {
   195→              setPhoneInput('');
   196→              setEditingPhone(true);
   197→            },
   198→          },
   199→        ]
   200→      );
   201→      return;
   202→    }
   203→    const phone = currentPhone.trim().replace(/[^0-9+]/g, '');
   204→    Linking.openURL(`sms:${phone}`);
   205→  }
   206→
   207→  function handleWhatsappPress() {
   208→    if (!shop) return;
   209→    if (!currentPhone) {
   210→      Alert.alert(
   211→        'No Phone Number',
   212→        "This shop doesn't have a phone number saved.",
   213→        [
   214→          { text: 'Cancel', style: 'cancel' },
   215→          {
   216→            text: 'Add Phone Number',
   217→            onPress: () => {
   218→              setPhoneInput('');
   219→              setEditingPhone(true);
   220→            },
   221→          },
   222→        ]
   223→      );
   224→      return;
   225→    }
   226→    let formattedPhone = currentPhone.trim().replace(/[^0-9]/g, '');
   227→    if (formattedPhone.startsWith('0')) {
   228→      formattedPhone = formattedPhone.substring(1);
   229→    }
   230→    if (!formattedPhone.startsWith('92')) {
   231→      formattedPhone = '92' + formattedPhone;
   232→    }
   233→    Linking.openURL(`https://wa.me/${formattedPhone}`);
   234→  }
   235→
   236→  async function handleSavePhone() {
   237→    if (!shop) return;
   238→    const trimmed = phoneInput.trim();
   239→    if (!trimmed) {
   240→      Alert.alert('Invalid Phone', 'Please enter a valid phone number.');
   241→      return;
   242→    }
   243→    setPhoneSaving(true);
   244→    try {
   245→      const trimmedOwner = ownerNameInput.trim();
   246→      await ApiService.updateShopPhone(shop.id, trimmed, trimmedOwner || undefined);
   247→      setLocalPhone(trimmed);
   248→      if (trimmedOwner) setLocalOwnerName(trimmedOwner);
   249→      setEditingPhone(false);
   250→      setPhoneInput('');
   251→      setOwnerNameInput('');
   252→      Alert.alert('Updated', 'Phone number and owner name have been updated successfully.');
   253→    } catch (e: any) {
   254→      Alert.alert('Error', e.message || 'Failed to update. Please try again.');
   255→    } finally {
   256→      setPhoneSaving(false);
   257→    }
   258→  }
   259→
   260→  function handleEditPhone() {
   261→    setPhoneInput(currentPhone);
   262→    setOwnerNameInput(currentOwnerName);
   263→    setEditingPhone(true);
   264→  }
   265→
   266→  if (!shop) return null;
   267→
   268→  const { balance: displayBalance, creditLimit: displayCreditLimit } = getShopDisplayBalance(shop, companyId);
   269→  const utilisationPct = displayCreditLimit > 0 ? Math.min((displayBalance / displayCreditLimit) * 100, 100) : 0;
   270→  const isOverLimit = displayBalance > displayCreditLimit;
   271→  const hasChartData = chartData.credits.some((v) => v > 0) || chartData.recoveries.some((v) => v > 0);
   272→
   273→  const totalRecovery = chartData.recoveries.reduce((s, v) => s + v, 0);
   274→  const totalCredit = chartData.credits.reduce((s, v) => s + v, 0);
   275→
   276→  return (
   277→    <Modal visible={visible} transparent animationType="slide" onRequestClose={onClose}>
   278→      <View style={styles.backdrop}>
   279→        <View style={styles.container}>
   280→          {/* Handle */}
   281→          <View style={styles.handle} />
   282→
   283→          {/* Gradient Header */}
   284→          <LinearGradient
   285→            colors={['#4F46E5', '#6366F1', '#818CF8']}
   286→            start={{ x: 0, y: 0 }}
   287→            end={{ x: 1, y: 1 }}
   288→            style={styles.header}
   289→          >
   290→            <View style={styles.headerBubble1} />
   291→            <View style={styles.headerBubble2} />
   292→            <View style={styles.shopIconWrap}>
   293→              <Text style={styles.shopIcon}>{shop.name.charAt(0).toUpperCase()}</Text>
   294→            </View>
   295→            <View style={styles.headerInfo}>
   296→              <Text style={styles.shopName} numberOfLines={1}>{shop.name}</Text>
   297→              <Text style={styles.owner}>{currentOwnerName || shop.ownerName}</Text>
   298→            </View>
   299→            <Pressable onPress={onClose} style={styles.closeBtn} hitSlop={12}>
   300→              <MaterialIcons name="close" size={22} color="rgba(255,255,255,0.8)" />
   301→            </Pressable>
   302→          </LinearGradient>
   303→
   304→          <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={styles.scroll}>
   305→            {/* Info chips */}
   306→            <View style={styles.chipRow}>
   307→              {shop.area ? (
   308→                <View style={styles.chip}>
   309→                  <MaterialIcons name="location-on" size={13} color={Colors.textSecondary} />
   310→                  <Text style={styles.chipText}>{shop.area}</Text>
   311→                </View>
   312→              ) : null}
   313→              {shop.routeDays && shop.routeDays.length > 0 ? (
   314→                <View style={[styles.chip, { backgroundColor: Colors.primaryLight }]}>
   315→                  <MaterialIcons name="calendar-today" size={13} color={Colors.primaryDark} />
   316→                  <Text style={[styles.chipText, { color: Colors.primaryDark }]}>
   317→                    {shop.routeDays.map(d => d.charAt(0).toUpperCase() + d.slice(1)).join(', ')}
   318→                  </Text>
   319→                </View>
   320→              ) : null}
   321→              {currentPhone ? (
   322→                <View style={styles.phoneChipRow}>
   323→                  <Pressable
   324→                    style={[styles.chip, { backgroundColor: '#EEF2FF' }]}
   325→                    onPress={() => Linking.openURL(`tel:${currentPhone}`)}
   326→                  >
   327→                    <MaterialIcons name="call" size={13} color="#4F46E5" />
   328→                    <Text style={[styles.chipText, { color: '#4F46E5' }]}>{currentPhone}</Text>
   329→                  </Pressable>
   330→                  <Pressable onPress={handleEditPhone} hitSlop={8} style={styles.phoneEditBtn}>
   331→                    <MaterialIcons name="edit" size={14} color="#4F46E5" />
   332→                  </Pressable>
   333→                </View>
   334→              ) : (
   335→                <Pressable
   336→                  style={[styles.chip, { backgroundColor: '#FEF3C7', borderColor: '#F59E0B', borderWidth: 1 }]}
   337→                  onPress={handleEditPhone}
   338→                >
   339→                  <MaterialIcons name="phone-disabled" size={13} color="#D97706" />
   340→                  <Text style={[styles.chipText, { color: '#D97706' }]}>No phone</Text>
   341→                  <MaterialIcons name="add-circle-outline" size={12} color="#D97706" />
   342→                </Pressable>
   343→              )}
   344→              {/* SMS & WhatsApp action chips */}
   345→              <Pressable
   346→                style={[styles.chip, { backgroundColor: '#E0F2FE' }]}
   347→                onPress={handleSmsPress}
   348→              >
   349→                <MaterialIcons name="sms" size={13} color="#0284C7" />
   350→                <Text style={[styles.chipText, { color: '#0284C7' }]}>SMS</Text>
   351→              </Pressable>
   352→              <Pressable
   353→                style={[styles.chip, { backgroundColor: '#DCFCE7' }]}
   354→                onPress={handleWhatsappPress}
   355→              >
   356→                <MaterialIcons name="chat" size={13} color="#16A34A" />
   357→                <Text style={[styles.chipText, { color: '#16A34A' }]}>WhatsApp</Text>
   358→              </Pressable>
   359→            </View>
   360→
   361→            {/* Balance + Credit cards */}
   362→            <View style={styles.balanceRow}>
   363→              <View style={styles.balanceCard}>
   364→                <View style={[styles.balanceIconWrap, { backgroundColor: Colors.dangerLight }]}>
   365→                  <MaterialIcons name="account-balance-wallet" size={20} color={Colors.danger} />
   366→                </View>
   367→                <Text style={styles.balanceLabel}>Outstanding</Text>
   368→                <Text style={[styles.balanceValue, { color: Colors.danger }]}>
   369→                  {formatPKR(displayBalance)}
   370→                </Text>
   371→              </View>
   372→              <View style={styles.balanceDivider} />
   373→              <View style={styles.balanceCard}>
   374→                <View style={[styles.balanceIconWrap, { backgroundColor: '#EEF2FF' }]}>
   375→                  <MaterialIcons name="credit-card" size={20} color="#4F46E5" />
   376→                </View>
   377→                <Text style={styles.balanceLabel}>Credit Limit</Text>
   378→                <Text style={[styles.balanceValue, { color: '#4F46E5' }]}>
   379→                  {formatPKR(displayCreditLimit)}
   380→                </Text>
   381→              </View>
   382→            </View>
   383→
   384→            {/* Credit utilisation */}
   385→            <View style={styles.utilisationCard}>
   386→              <View style={styles.utilisationHeader}>
   387→                <Text style={styles.utilisationLabel}>Credit Utilisation</Text>
   388→                <Text style={[styles.utilisationPct, { color: isOverLimit ? Colors.danger : utilisationPct > 80 ? Colors.secondary : Colors.primary }]}>
   389→                  {utilisationPct.toFixed(0)}%
   390→                </Text>
   391→              </View>
   392→              <View style={styles.progressTrack}>
   393→                <View
   394→                  style={[
   395→                    styles.progressFill,
   396→                    {
   397→                      width: `${utilisationPct}%`,
   398→                      backgroundColor: isOverLimit ? Colors.danger : utilisationPct > 80 ? Colors.secondary : Colors.primary,
   399→                    },
   400→                  ]}
   401→                />
   402→              </View>
   403→              {isOverLimit ? (
   404→                <View style={styles.overLimitRow}>
   405→                  <MaterialIcons name="warning" size={14} color={Colors.danger} />
   406→                  <Text style={styles.overLimitText}>Over Credit Limit by {formatPKR(displayBalance - displayCreditLimit)}</Text>
   407→                </View>
   408→              ) : null}
   409→            </View>
   410→
   411→            {/* 6-Month Performance Chart */}
   412→            <View style={styles.chartCard}>
   413→              <View style={styles.chartTitleRow}>
   414→                <View>
   415→                  <Text style={styles.sectionTitle}>6-Month Performance</Text>
   416→                  <Text style={styles.chartSubtitle}>Credit vs Recovery trend</Text>
   417→                </View>
   418→                <View style={styles.chartLegend}>
   419→                  <View style={styles.legendItem}>
   420→                    <View style={[styles.legendDot, { backgroundColor: Colors.secondary }]} />
   421→                    <Text style={styles.legendText}>Credit</Text>
   422→                  </View>
   423→                  <View style={styles.legendItem}>
   424→                    <View style={[styles.legendDot, { backgroundColor: Colors.primary }]} />
   425→                    <Text style={styles.legendText}>Recovery</Text>
   426→                  </View>
   427→                </View>
   428→              </View>
   429→
   430→              {/* Summary pills */}
   431→              <View style={styles.chartPills}>
   432→                <View style={[styles.chartPill, { backgroundColor: Colors.secondaryLight }]}>
   433→                  <Text style={styles.chartPillLabel}>Total Credit</Text>
   434→                  <Text style={[styles.chartPillValue, { color: Colors.secondary }]}>{formatPKR(totalCredit)}</Text>
   435→                </View>
   436→                <View style={[styles.chartPill, { backgroundColor: Colors.primaryLight }]}>
   437→                  <Text style={styles.chartPillLabel}>Total Recovery</Text>
   438→                  <Text style={[styles.chartPillValue, { color: Colors.primary }]}>{formatPKR(totalRecovery)}</Text>
   439→                </View>
   440→              </View>
   441→
   442→              {chartLoading ? (
   443→                <View style={styles.chartLoading}>
   444→                  <ActivityIndicator size="small" color={Colors.primary} />
   445→                  <Text style={styles.chartLoadingText}>Loading chart...</Text>
   446→                </View>
   447→              ) : !hasChartData ? (
   448→                <View style={styles.chartEmpty}>
   449→                  <Text style={styles.chartEmptyIcon}>📊</Text>
   450→                  <Text style={styles.chartEmptyText}>No transaction history available</Text>
   451→                </View>
   452→              ) : (
   453→                <BarChart
   454→                  data={{
   455→                    labels: chartData.labels,
   456→                    datasets: [
   457→                      {
   458→                        data: chartData.recoveries.map((v) => Math.max(v, 0)),
   459→                        color: (opacity = 1) => `rgba(79, 70, 229, ${opacity})`,
   460→                      },
   461→                    ],
   462→                  }}
   463→                  width={screenWidth - 80}
   464→                  height={150}
   465→                  yAxisLabel="Rs."
   466→                  yAxisSuffix=""
   467→                  chartConfig={{
   468→                    backgroundColor: Colors.surface,
   469→                    backgroundGradientFrom: Colors.surface,
   470→                    backgroundGradientTo: Colors.surface,
   471→                    decimalPlaces: 0,
   472→                    color: (opacity = 1) => `rgba(79, 70, 229, ${opacity})`,
   473→                    labelColor: (opacity = 1) => `rgba(107, 114, 128, ${opacity})`,
   474→                    barPercentage: 0.7,
   475→                    propsForBackgroundLines: {
   476→                      stroke: Colors.borderLight,
   477→                    },
   478→                  }}
   479→                  style={{ borderRadius: Radius.md, marginLeft: -Spacing.md }}
   480→                  showValuesOnTopOfBars={false}
   481→                  withInnerLines
   482→                  fromZero
   483→                />
   484→              )}
   485→            </View>
   486→
   487→            {/* Recent transactions */}
   488→            <Text style={styles.sectionTitle}>Recent Transactions</Text>
   489→
   490→            {/* Notes Section */}
   491→            <View style={styles.notesSection}>
   492→              <View style={styles.notesHeader}>
   493→                <View style={styles.notesHeaderLeft}>
   494→                  <View style={styles.notesIconWrap}>
   495→                    <MaterialIcons name="sticky-note-2" size={16} color={Colors.secondary} />
   496→                  </View>
   497→                  <Text style={styles.sectionTitleInline}>Notes / Remarks</Text>
   498→                </View>
   499→                {shopNote ? (
   500→                  <Pressable onPress={handleDeleteNote} hitSlop={8} style={styles.noteDeleteBtn}>
   501→                    <MaterialIcons name="delete-outline" size={16} color={Colors.danger} />
   502→                  </Pressable>
   503→                ) : null}
   504→              </View>
   505→              <View style={styles.noteInputWrap}>
   506→                <TextInput
   507→                  style={styles.noteTextInput}
   508→                  value={noteInput}
   509→                  onChangeText={setNoteInput}
   510→                  placeholder="Add a note about this shop..."
   511→                  placeholderTextColor={Colors.textMuted}
   512→                  maxLength={500}
   513→                  multiline
   514→                  numberOfLines={3}
   515→                />
   516→              </View>
   517→              <View style={styles.noteActions}>
   518→                {shopNote ? (
   519→                  <Text style={styles.noteUpdatedAt}>
   520→                    Last updated: {formatDateTime(shopNote.updatedAt)}
   521→                  </Text>
   522→                ) : null}
   523→                <Pressable
   524→                  style={({ pressed }) => [styles.noteSaveBtn, pressed && { opacity: 0.7 }]}
   525→                  onPress={handleSaveNote}
   526→                  disabled={!noteInput.trim() || noteSaving}
   527→                >
   528→                  {noteSaving ? (
   529→                    <ActivityIndicator size="small" color="#FFFFFF" />
   530→                  ) : (
   531→                    <MaterialIcons name="save" size={14} color="#FFFFFF" />
   532→                  )}
   533→                  <Text style={styles.noteSaveBtnText}>
   534→                    {noteSaving ? 'Saving...' : shopNote ? 'Update' : 'Save Note'}
   535→                  </Text>
   536→                </Pressable>
   537→              </View>
   538→            </View>
   539→            {loading ? (
   540→              <ActivityIndicator color={Colors.primary} style={{ marginVertical: Spacing.md }} />
   541→            ) : recentTxns.length === 0 ? (
   542→              <View style={styles.emptyTxnWrap}>
   543→                <MaterialIcons name="receipt-long" size={36} color={Colors.textMuted} />
   544→                <Text style={styles.emptyTxn}>No transactions found</Text>
   545→              </View>
   546→            ) : (
   547→              recentTxns.map((txn) => (
   548→                <View key={txn.id} style={styles.txnRow}>
   549→                  <View style={[
   550→                    styles.txnTypeIcon,
   551→                    { backgroundColor: txn.type === 'credit' ? Colors.secondaryLight : Colors.primaryLight },
   552→                  ]}>
   553→                    <MaterialIcons
