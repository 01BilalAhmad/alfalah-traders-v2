   135→function ActionRow({
   136→  icon,
   137→  iconColor,
   138→  iconBg,
   139→  title,
   140→  subtitle,
   141→  onPress,
   142→  showChevron = true,
   143→  danger = false,
   144→}: {
   145→  icon: string;

   147→  iconBg: string;
   148→  title: string;
   149→  subtitle: string;
   150→  onPress: () => void;
   151→  showChevron?: boolean;
   152→  danger?: boolean;
   153→}) {
   154→  return (
   155→    <Pressable
   156→      style={({ pressed }) => [actionStyles.card, pressed && { opacity: 0.9, transform: [{ scale: 0.995}] }]}

   159→      <View style={[actionStyles.iconWrap, { backgroundColor: iconBg }]}>

   164→        <Text style={actionStyles.subtitle}>{subtitle}</Text>
   165→      </View>
   166→      {showChevron ? (
   167→        <MaterialIcons name="chevron-right" size={22} color={Colors.textMuted} />
   168→      ) : null}
   169→    </Pressable>
   170→  );
   171→}

   208→export default function ProfileScreen() {
   209→  const insets = useSafeAreaInsets();
   210→  const router = useRouter();
   211→  const { user, logout, updatePhone, companies, selectedCompanyId } = useAuth();
   212→  const { allShops } = useShops();
   213→  const { setNeedsPinSetup, lock } = useLock();
   214→  const { isTracking, isStarting, isStopping, startRoute, endRoute, startTime, error: routeError } = useRouteTracking();
   215→
   216→  const [loggingOut, setLoggingOut] = useState(false);
   217→  const [todayRecovery, setTodayRecovery] = useState(0);
   218→  const [thisMonthRecovery, setThisMonthRecovery] = useState(0);
   219→  const [showAnalysis, setShowAnalysis] = useState(false);
   220→
   221→  // Phone edit state
   222→  const [isEditingPhone, setIsEditingPhone] = useState(false);
   223→  const [phoneInput, setPhoneInput] = useState('');
   224→  const [isSavingPhone, setIsSavingPhone] = useState(false);
   225→  const phoneInputRef = useRef<TextInput>(null);
   226→
   227→  useEffect(() => {
   228→    if (user) loadStats();
   229→  }, [user]);
   230→
   231→  async function loadStats() {
   232→    if (!user) return;
   233→    try {
   234→      const today = await ApiService.getRecoverySummary(getTodayDateStr());
   235→      const myToday = today.orderbookers.find((ob) => ob.orderbookerId === user.id);
   236→      if (myToday) setTodayRecovery(myToday.totalRecovery);
   237→
   238→      const now = new Date();
   239→      const firstOfMonth = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-01`;
   240→      const monthTxns = await ApiService.getTransactions({
   241→        createdBy: user.id,
   242→        type: 'recovery',
   243→        date: firstOfMonth,
   244→        limit: 500,
   245→      });
   246→      setThisMonthRecovery(monthTxns.transactions.reduce((sum, t) => sum + t.amount, 0));
   247→    } catch { /* not critical */ }
   248→  }
   249→
   250→  const handleSavePhone = async () => {
   251→    const trimmed = phoneInput.trim();
   252→    if (trimmed && !/^[\d+\-\s()]{7,15}$/.test(trimmed)) {
   253→      Alert.alert('Invalid Phone', 'Please enter a valid phone number (7-15 digits)');
   254→      return;
   255→    }
   256→    setIsSavingPhone(true);
   257→    try {
   258→      await updatePhone(trimmed);
   259→      setIsEditingPhone(false);
   260→      setPhoneInput('');
   261→      Alert.alert('Phone Updated', trimmed ? `Phone number updated to ${trimmed}` : 'Phone number removed');
   262→    } catch (err: any) {
   263→      Alert.alert('Error', err?.message || 'Failed to update phone number');
   264→    } finally {
   265→      setIsSavingPhone(false);
   266→    }
   267→  };
   268→
   269→  const handleLogout = () => {
   270→    // If route is active, warn user first
   271→    if (isTracking) {
   272→      Alert.alert(
   273→        'Active Route Detected',
   274→        'You have an active route session. Logging out will stop route tracking. Do you want to end the route and logout?',
   275→        [
   276→          { text: 'Cancel', style: 'cancel' },
   277→          {
   278→            text: 'End Route & Logout',
   279→            style: 'destructive',
   280→            onPress: async () => {
   281→              setLoggingOut(true);
   282→              try {
   283→                await endRoute();
   284→              } catch {}
   285→              try {
   286→                await logout();
   287→                await SecureStorageService.clearAll();
   288→                setTimeout(() => {
   289→                  router.replace('/');
   290→                }, 100);
   291→              } finally {
   292→                setLoggingOut(false);
   293→              }
   294→            },
   295→          },
   296→        ]
   297→      );
   298→      return;
   299→    }
   300→
   301→    Alert.alert('Logout', 'Are you sure you want to logout?', [
   302→      { text: 'Cancel', style: 'cancel' },
   303→      {
   304→        text: 'Logout',
   305→        style: 'destructive',
   306→        onPress: async () => {
   307→          setLoggingOut(true);
   308→          try {
   309→            await logout();
   310→            await SecureStorageService.clearAll();
   311→            router.replace('/login' as any);
   312→          } finally {
   313→            setLoggingOut(false);
   314→          }
   315→        },
   316→      },
   317→    ]);
   318→  };
   319→
   320→  const handleChangePin = () => {
   321→    Alert.alert('Change PIN', 'Do you want to change your PIN? You will need to set a new one.', [
   322→      { text: 'Cancel', style: 'cancel' },
   323→      {
   324→        text: 'Change PIN',
   325→        onPress: async () => {
   326→          await SecureStorageService.clearPin();
   327→          setNeedsPinSetup(true);
   328→          lock();
   329→        },
   330→      },
   331→    ]);
   332→  };
   333→
   334→  if (!user) return null;
   335→
   336→  const initials = user.name
   337→    .split(' ')
   338→    .map((w) => w[0])
   339→    .slice(0, 2)
   340→    .join('')
   341→    .toUpperCase();
   342→
   343→  return (
   344→    <View style={[styles.root, { paddingTop: insets.top }]}>
   345→      <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={styles.scroll}>
   346→        {/* ── Hero profile card — 3-color gradient ── */}
   347→        <LinearGradient
   348→          colors={['#4F46E5', '#6366F1', '#818CF8']}
   349→          start={{ x: 0, y: 0 }}
   350→          end={{ x: 1, y: 1 }}
   351→          style={styles.profileHero}
   352→        >
   353→          {/* Decorative circles */}
   354→          <View style={styles.bubble1} />
   355→          <View style={styles.bubble2} />
   356→          <View style={styles.bubble3} />
   357→
   358→          {/* Rounded Square Avatar */}
   359→          <View style={styles.avatarWrap}>
   360→            <View style={styles.avatarSquare}>
   361→              <Text style={styles.avatarText}>{initials}</Text>
   362→            </View>
   363→            <View style={styles.activeIndicator} />
   364→          </View>
   365→          <Text style={styles.profileName}>{user.name}</Text>
   366→          <View style={styles.roleBadge}>
   367→            <MaterialIcons name="badge" size={12} color="rgba(255,255,255,0.9)" />
   368→            <Text style={styles.roleText}>ORDER BOOKER</Text>
   369→          </View>
   370→          {companies.length > 1 && selectedCompanyId ? (
   371→            <View style={styles.companyBadge}>
   372→              <MaterialIcons name="business" size={11} color="rgba(255,255,255,0.8)" />
   373→              <Text style={styles.companyBadgeText}>
   374→                {companies.find((c) => c.companyId === selectedCompanyId)?.companyName || user.companyName || ''}
   375→              </Text>
   376→            </View>
   377→          ) : null}
   378→        </LinearGradient>
   379→
   380→        {/* ── Account Details Card ── */}
   381→        <View style={styles.card}>
   382→          <View style={styles.cardHeader}>
   383→            <View style={styles.cardHeaderIcon}>
   384→              <MaterialIcons name="person-outline" size={16} color="#4F46E5" />
   385→            </View>
   386→            <Text style={styles.cardTitle}>Account Details</Text>
   387→          </View>
   388→          <InfoRow icon="person" label="Username" value={`@${user.username}`} />
   389→          <View style={styles.divider} />
   390→          {/* Phone with edit */}
   391→          <View style={infoRowStyles.row}>
   392→            <View style={infoRowStyles.iconWrap}>
   393→              <MaterialIcons name="call" size={18} color="#4F46E5" />
   394→            </View>
   395→            <View style={{ flex: 1 }}>
   396→              <Text style={infoRowStyles.label}>Phone Number</Text>
   397→              {isEditingPhone ? (
   398→                <View style={{ flexDirection: 'row', alignItems: 'center', gap: 8, marginTop: 4 }}>
   399→                  <TextInput
   400→                    ref={phoneInputRef}
   401→                    value={phoneInput}
   402→                    onChangeText={setPhoneInput}
   403→                    placeholder="03XXXXXXXXX"
   404→                    keyboardType="phone-pad"
   405→                    maxLength={15}
   406→                    style={{
   407→                      flex: 1,
   408→                      height: 36,
   409→                      borderWidth: 1,
   410→                      borderColor: Colors.primary,
   411→                      borderRadius: Radius.sm,
   412→                      paddingHorizontal: 10,
   413→                      fontSize: FontSize.base,
   414→                      color: Colors.text,
   415→                      backgroundColor: Colors.background,
   416→                    }}
   417→                    autoFocus
   418→                    editable={!isSavingPhone}
   419→                    onSubmitEditing={handleSavePhone}
   420→                  />
   421→                  <Pressable
   422→                    onPress={handleSavePhone}
   423→                    disabled={isSavingPhone}
   424→                    style={{
   425→                      width: 32,
   426→                      height: 32,
   427→                      borderRadius: 16,
   428→                      backgroundColor: Colors.primary,
   429→                      alignItems: 'center',
   430→                      justifyContent: 'center',
   431→                    }}
   432→                  >
   433→                    {isSavingPhone ? (
   434→                      <ActivityIndicator size="small" color="#FFFFFF" />
   435→                    ) : (
   436→                      <MaterialIcons name="check" size={18} color="#FFFFFF" />
   437→                    )}
   438→                  </Pressable>
   439→                  <Pressable
   440→                    onPress={() => { setIsEditingPhone(false); setPhoneInput(''); }}
   441→                    disabled={isSavingPhone}
   442→                    style={{
   443→                      width: 32,
   444→                      height: 32,
   445→                      borderRadius: 16,
   446→                      backgroundColor: Colors.borderLight,
   447→                      alignItems: 'center',
   448→                      justifyContent: 'center',
   449→                    }}
   450→                  >
   451→                    <MaterialIcons name="close" size={18} color={Colors.textSecondary} />
   452→                  </Pressable>
   453→                </View>
   454→              ) : (
   455→                <View style={{ flexDirection: 'row', alignItems: 'center', gap: 6 }}>
   456→                  <Text style={infoRowStyles.value}>{user.phone || 'Not set'}</Text>
   457→                  <Pressable
   458→                    onPress={() => {
   459→                      setPhoneInput(user.phone || '');
   460→                      setIsEditingPhone(true);
   461→                      setTimeout(() => phoneInputRef.current?.focus(), 100);
   462→                    }}
   463→                    style={{
   464→                      width: 24,
   465→                      height: 24,
   466→                      borderRadius: 12,
   467→                      backgroundColor: Colors.primaryLight,
   468→                      alignItems: 'center',
   469→                      justifyContent: 'center',
   470→                    }}
   471→                  >
   472→                    <MaterialIcons name="edit" size={14} color={Colors.primary} />
   473→                  </Pressable>
   474→                </View>
   475→              )}
   476→            </View>
   477→          </View>
   478→          <View style={styles.divider} />
   479→          <InfoRow
   480→            icon="verified-user"
   481→            label="Account Status"
   482→            value={user.status === 'active' ? 'Active' : user.status}
   483→            valueColor={user.status === 'active' ? '#10B981' : undefined}
   484→          />
   485→          {user.companyName || selectedCompanyId ? (
   486→            <>
   487→              <View style={styles.divider} />
   488→              <InfoRow
   489→                icon="business"
   490→                label="Company"
   491→                value={selectedCompanyId
   492→                  ? companies.find((c) => c.companyId === selectedCompanyId)?.companyName || user.companyName || 'N/A'
   493→                  : user.companyName || 'N/A'
   494→                }
   495→              />
   496→            </>
   497→          ) : null}
   498→        </View>
   499→
   500→        {/* ── Performance KPIs ── */}
   501→        <View style={styles.sectionHeader}>
   502→          <View style={styles.sectionHeaderIcon}>
   503→            <MaterialIcons name="analytics" size={16} color="#4F46E5" />
   504→          </View>
   505→          <Text style={styles.sectionTitle}>Performance Overview</Text>
   506→        </View>
   507→        <View style={styles.kpiGrid}>
   508→          <KpiCard
   509→            label="Month Recovery"
   510→            value={
   511→              thisMonthRecovery >= 1000000
   512→                ? `Rs. ${(thisMonthRecovery / 1000000).toFixed(1)}M`
   513→                : thisMonthRecovery >= 1000
   514→                ? `Rs. ${(thisMonthRecovery / 1000).toFixed(0)}K`
   515→                : formatPKR(thisMonthRecovery)
   516→            }
   517→            color={Colors.primary}
   518→            bg={Colors.primaryLight}
   519→            icon="trending-up"
   520→          />
   521→          <KpiCard
   522→            label="Today's Recovery"
   523→            value={
   524→              todayRecovery >= 1000
   525→                ? `Rs. ${(todayRecovery / 1000).toFixed(0)}K`
   526→                : formatPKR(todayRecovery)
   527→            }
   528→            color="#3B82F6"
   529→            bg="#EFF6FF"
   530→            icon="today"
   531→            subtext="Pending approval"
   532→          />
   533→          <KpiCard
   534→            label="Shops Assigned"
   535→            value={String(allShops.length)}
   536→            color={Colors.secondary}
   537→            bg={Colors.secondaryLight}
   538→            icon="store"
   539→          />
   540→        </View>
   541→
   542→        {/* Recovery Comparison: This vs Last Week */}
   543→        <RecoveryComparison userId={user.id} />
   544→
   545→        {/* ── Recovery Analysis Toggle ── */}
   546→        <Pressable
   547→          style={({ pressed }) => [styles.analysisToggle, pressed && { opacity: 0.9, transform: [{ scale: 0.995 }] }]}
   548→          onPress={() => setShowAnalysis((v) => !v)}
   549→        >
   550→          <View style={styles.analysisToggleLeft}>
   551→            <View style={styles.analysisToggleIcon}>
   552→              <MaterialIcons name="analytics" size={20} color="#4F46E5" />
   553→            </View>
   554→            <View>
   555→              <Text style={styles.analysisToggleTitle}>Recovery Analysis</Text>
   556→              <Text style={styles.analysisToggleSub}>Credit vs recovery chart</Text>
   557→            </View>
   558→          </View>
   559→          <View style={[styles.toggleIconWrap, showAnalysis && styles.toggleIconWrapActive]}>
   560→            <MaterialIcons
   561→              name={showAnalysis ? 'keyboard-arrow-up' : 'keyboard-arrow-down'}
   562→              size={22}
   563→              color={showAnalysis ? '#4F46E5' : Colors.textSecondary}
   564→            />
   565→          </View>
   566→        </Pressable>
   567→
   568→        {showAnalysis ? <RecoveryAnalysisChart userId={user.id} /> : null}
   569→
   570→        {/* Performance Ranking */}
   571→        <PerformanceRanking />
   572→
   573→        {/* ── Route Tracking Section ── */}
   574→        <View style={styles.sectionHeader}>
   575→          <View style={styles.sectionHeaderIcon}>
   576→            <MaterialIcons name="navigation" size={16} color="#4F46E5" />
   577→          </View>
   578→          <Text style={styles.sectionTitle}>Route Tracking</Text>
   579→        </View>
   580→
   581→        {routeError ? (
   582→          <View style={{
   583→            marginHorizontal: Spacing.md,
   584→            marginBottom: Spacing.sm,
   585→            backgroundColor: '#FEF2F2',
   586→            borderRadius: 12,
   587→            padding: 12,
   588→            borderWidth: 1,
   589→            borderColor: '#FECACA',
   590→          }}>
   591→            <Text style={{ fontSize: FontSize.xs, color: '#DC2626' }}>{routeError}</Text>
   592→          </View>
   593→        ) : null}
   594→
   595→        {isTracking ? (
   596→          <View style={{
   597→            marginHorizontal: Spacing.md,
   598→            marginBottom: Spacing.sm,
   599→            backgroundColor: '#ECFDF5',
   600→            borderRadius: 16,
