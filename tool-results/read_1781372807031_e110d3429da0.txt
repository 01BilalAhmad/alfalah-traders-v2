   111→function StreakBadge({ streak }: { streak: number }) {
   112→  if (streak >= 7) {
   113→    return (
   114→      <Badge className="text-[10px] bg-orange-100 text-orange-700 border-orange-200 dark:bg-orange-900/50 dark:text-orange-300 dark:border-orange-800 font-semibold">
   115→        <Flame className="h-3 w-3 mr-1" /> {streak} days — On Fire!
   116→      </Badge>
   117→    );
   118→  }
   119→  if (streak >= 3) {
   120→    return (
   121→      <Badge className="text-[10px] bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/50 dark:text-amber-300 dark:border-amber-800 font-semibold">
   122→        {streak} days — Good
   123→      </Badge>
   124→    );
   125→  }
   126→  if (streak >= 1) {
   127→    return (
   128→      <Badge className="text-[10px] bg-blue-100 text-blue-700 border-blue-200 dark:bg-blue-900/50 dark:text-blue-300 dark:border-blue-800 font-semibold">
   129→        {streak} day{streak > 1 ? 's' : ''}
   130→      </Badge>
   131→    );
   132→  }
   133→  return (
   134→    <Badge variant="outline" className="text-[10px]">Inactive</Badge>
   135→  );
   136→}

   138→export default function AdminVisitTracking() {
   139→  const { setSelectedShopId, setSelectedShopName } = useAppStore();
   140→  const router = useRouter();
   141→  const [orderbookers, setOrderbookers] = useState<Orderbooker[]>([]);
   142→  const [streaks, setStreaks] = useState<Record<string, VisitStreak>>({});

   144→  const [loading, setLoading] = useState(true);
   145→  const [selectedOB, setSelectedOB] = useState<string>('all');
   146→  const [searchQuery, setSearchQuery] = useState('');
   147→  const [visitsLoading, setVisitsLoading] = useState(false);
   148→
   149→  const fetchStreaks = useCallback(async () => {
   150→    setLoading(true);
   151→    try {
   152→      const obRes = await apiFetch('/api/orderbookers');
   153→      if (!obRes.ok) {
   154→        toast({ title: 'Error', description: 'Failed to load orderbookers', variant: 'destructive' });
   155→        setLoading(false);
   156→        return;
   157→      }
   158→      const obs = await obRes.json();
   159→      const activeOBs = Array.isArray(obs) ? obs.filter((o: Orderbooker) => o.status === 'active') : [];
   160→      setOrderbookers(activeOBs);
   161→
   162→      // Fetch streaks for each OB
   163→      const streakPromises = activeOBs.map((ob: Orderbooker) =>
   164→        apiFetch(`/api/users/${ob.id}/visit-streak`)
   165→          .then(r => r.ok ? r.json() : null)
   166→          .catch(() => null)
   167→      );
   168→      const streakResults = await Promise.all(streakPromises);
   169→      const streakMap: Record<string, VisitStreak> = {};
   170→      streakResults.forEach((s, i) => {
   171→        if (s) streakMap[activeOBs[i].id] = s;
   172→      });
   173→      setStreaks(streakMap);
   174→    } catch {
   175→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' });
   176→    } finally {
   177→      setLoading(false);
   178→    }
   179→  }, []);
   180→

   182→    setVisitsLoading(true);
   183→    try {
   184→      // Use the global visits endpoint that combines ShopVisit + Transaction data
   185→      const res = await apiFetch('/api/visits/recent?limit=200');
   186→      if (!res.ok) {
   187→        setVisitsLoading(false);
   188→        return;
   189→      }
   190→      const visits = await res.json();
   191→      setRecentVisits(Array.isArray(visits) ? visits : []);
   192→    } catch {
   193→      // silent
   194→    } finally {
   195→      setVisitsLoading(false);
   196→    }
   197→  }, []);
   198→
   199→  useEffect(() => {
   200→    fetchStreaks();
   201→    fetchRecentVisits();
   202→
   203→    // Auto-refresh every 30 seconds so admin sees new visits in real-time
   204→    const interval = setInterval(() => {
   205→      fetchRecentVisits();
   206→    }, 30000);
   207→
   208→    return () => clearInterval(interval);
   209→  }, [fetchRecentVisits]);
   210→
   211→  // Filtered visits
   212→  const filteredVisits = useMemo(() => {
   213→    let result = recentVisits;
   214→    if (selectedOB !== 'all') {
   215→      result = result.filter(v => v.orderbookerId === selectedOB);
   216→    }
   217→    if (searchQuery.trim()) {
   218→      const q = searchQuery.toLowerCase();
   219→      result = result.filter(
   220→        v => v.shopName?.toLowerCase().includes(q) ||
   221→             v.orderbookerName?.toLowerCase().includes(q) ||
   222→             v.gpsAddress?.toLowerCase().includes(q)
   223→      );
   224→    }
   225→    return result;
   226→  }, [recentVisits, selectedOB, searchQuery]);
   227→
   228→  // Summary
   229→  const summary = useMemo(() => {
   230→    const streakValues = Object.values(streaks);
   231→    const totalVisitDays = streakValues.reduce((s, v) => s + v.totalVisitDays, 0);
   232→    const avgStreak = streakValues.length > 0
   233→      ? Math.round(streakValues.reduce((s, v) => s + v.currentStreak, 0) / streakValues.length)
   234→      : 0;
   235→    const bestStreak = Math.max(0, ...streakValues.map(v => v.currentStreak));
   236→
   237→    // Active today: count OBs who have visits in recentVisits (today's data)
   238→    const obsWithVisitToday = new Set(recentVisits.map(v => v.orderbookerId));
   239→    // Also check streak data
   240→    const obsWithStreakToday = streakValues.filter(v => {
   241→      if (!v.lastVisitDate) return false;
   242→      const today = new Date().toLocaleDateString('en-CA', { timeZone: 'Asia/Karachi' });
   243→      return v.lastVisitDate.startsWith(today);
   244→    }).map(v => v.orderbookerId);
   245→    obsWithStreakToday.forEach(id => obsWithVisitToday.add(id));
   246→
   247→    const activeToday = obsWithVisitToday.size;
   248→
   249→    // Shops visited today count
   250→    const shopsVisitedToday = new Set(recentVisits.map(v => v.shopId)).size;
   251→
   252→    return { totalVisitDays, avgStreak, bestStreak, activeToday, totalOBs: orderbookers.length, shopsVisitedToday };
   253→  }, [streaks, orderbookers, recentVisits]);
   254→

   257→    setSelectedShopName(shopName);
   258→    router.push(`/shops/${shopId}`);
   259→  };

   261→  if (loading) return <VisitSkeleton />;
   262→

   264→    <div className="space-y-6">

   266→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   267→        <div>
   268→          <h2 className="text-lg font-bold flex items-center gap-2">
   269→            <Navigation className="h-5 w-5 text-primary" />
   270→            Visit Tracking
   271→          </h2>
   272→          <p className="text-sm text-muted-foreground mt-1">
   273→            Today's shop visits by orderbookers (from recovery, credit & GPS check-ins)
   274→          </p>
   275→        </div>
   276→        <div className="flex items-center gap-3">
   277→          <Button size="sm" variant="outline" className="h-9 gap-1.5" onClick={() => { fetchStreaks(); fetchRecentVisits(); }}>
   278→            <RefreshCw className="h-3.5 w-3.5" />
   279→            Refresh
   280→          </Button>
   281→        </div>
   282→      </div>
   283→
   284→      {/* Summary KPIs */}
   285→      <div className="grid grid-cols-2 lg:grid-cols-5 gap-4 stagger-children">
   286→        <Card className="card-elevated card-hover border border-border">
   287→          <CardContent className="p-4">
   288→            <div className="flex items-center justify-between mb-3">
   289→              <div className="h-10 w-10 rounded-xl bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center shadow-sm">
   290→                <Flame className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   291→              </div>
   292→              <Badge variant="secondary" className="text-[10px] font-medium">Best</Badge>
   293→            </div>
   294→            <p className="text-xs text-muted-foreground font-medium mb-0.5">Best Active Streak</p>
   295→            <p className="text-2xl font-bold text-foreground tabular-nums number-animate">{summary.bestStreak}d</p>
   296→          </CardContent>
   297→        </Card>
   298→        <Card className="card-elevated card-hover border border-border">
   299→          <CardContent className="p-4">
   300→            <div className="flex items-center justify-between mb-3">
   301→              <div className="h-10 w-10 rounded-xl bg-indigo-100 dark:bg-indigo-900/40 flex items-center justify-center shadow-sm">
   302→                <Users className="h-5 w-5 text-indigo-600 dark:text-indigo-400" />
   303→              </div>
   304→              <Badge variant="secondary" className="text-[10px] font-medium">Average</Badge>
   305→            </div>
   306→            <p className="text-xs text-muted-foreground font-medium mb-0.5">Avg Active Streak</p>
   307→            <p className="text-2xl font-bold text-foreground tabular-nums number-animate">{summary.avgStreak}d</p>
   308→          </CardContent>
   309→        </Card>
   310→        <Card className="card-elevated card-hover border border-border">
   311→          <CardContent className="p-4">
   312→            <div className="flex items-center justify-between mb-3">
   313→              <div className="h-10 w-10 rounded-xl bg-violet-100 dark:bg-violet-900/40 flex items-center justify-center shadow-sm">
   314→                <Calendar className="h-5 w-5 text-violet-600 dark:text-violet-400" />
   315→              </div>
   316→              <Badge variant="secondary" className="text-[10px] font-medium">All Time</Badge>
   317→            </div>
   318→            <p className="text-xs text-muted-foreground font-medium mb-0.5">Total Visit Days</p>
   319→            <p className="text-2xl font-bold text-foreground tabular-nums number-animate">{summary.totalVisitDays}</p>
   320→          </CardContent>
   321→        </Card>
   322→        <Card className="card-elevated card-hover border border-border">
   323→          <CardContent className="p-4">
   324→            <div className="flex items-center justify-between mb-3">
   325→              <div className="h-10 w-10 rounded-xl bg-rose-100 dark:bg-rose-900/40 flex items-center justify-center shadow-sm">
   326→                <MapPin className="h-5 w-5 text-rose-600 dark:text-rose-400" />
   327→              </div>
   328→              <Badge variant="secondary" className="text-[10px] font-medium">Today</Badge>
   329→            </div>
   330→            <p className="text-xs text-muted-foreground font-medium mb-0.5">Shops Visited Today</p>
   331→            <p className="text-2xl font-bold text-foreground tabular-nums number-animate">
   332→              {summary.shopsVisitedToday}
   333→            </p>
   334→          </CardContent>
   335→        </Card>
   336→        <Card className="card-elevated card-hover border border-border">
   337→          <CardContent className="p-4">
   338→            <div className="flex items-center justify-between mb-3">
   339→              <div className="h-10 w-10 rounded-xl bg-blue-100 dark:bg-blue-900/40 flex items-center justify-center shadow-sm">
   340→                <CheckCircle2 className="h-5 w-5 text-blue-600 dark:text-blue-400" />
   341→              </div>
   342→              <Badge variant="secondary" className="text-[10px] font-medium">Today</Badge>
   343→            </div>
   344→            <p className="text-xs text-muted-foreground font-medium mb-0.5">OBs Active Today</p>
   345→            <p className="text-2xl font-bold text-foreground tabular-nums number-animate">
   346→              {summary.activeToday}/{summary.totalOBs}
   347→            </p>
   348→          </CardContent>
   349→        </Card>
   350→      </div>
   351→
   352→      {/* OB Streaks Table */}
   353→      <Card className="card-elevated">
   354→        <CardHeader className="pb-3 pt-4 px-5">
   355→          <CardTitle className="text-base font-semibold flex items-center gap-2">
   356→            <Flame className="h-4 w-4 text-primary" />
   357→            Orderbooker Visit Streaks
   358→          </CardTitle>
   359→        </CardHeader>
   360→        <CardContent className="p-0">
   361→          <div className="overflow-x-auto">
   362→            <Table>
   363→              <TableHeader>
   364→                <TableRow className="bg-primary hover:bg-transparent">
   365→                  <TableHead className="text-white font-semibold text-xs w-12">#</TableHead>
   366→                  <TableHead className="text-white font-semibold text-xs">Name</TableHead>
   367→                  <TableHead className="text-white font-semibold text-xs text-center">Current Streak</TableHead>
   368→                  <TableHead className="text-white font-semibold text-xs text-center hidden sm:table-cell">Longest Streak</TableHead>
   369→                  <TableHead className="text-white font-semibold text-xs text-center hidden md:table-cell">Total Visit Days</TableHead>
   370→                  <TableHead className="text-white font-semibold text-xs text-center hidden lg:table-cell">Last Visit</TableHead>
   371→                </TableRow>
   372→              </TableHeader>
   373→              <TableBody>
   374→                {orderbookers.length === 0 ? (
   375→                  <TableRow>
   376→                    <TableCell colSpan={6}>
   377→                      <div className="text-center py-10">
   378→                        <AlertCircle className="h-10 w-10 mx-auto mb-2 text-muted-foreground/40" />

   381→                    </TableCell>

   385→                    .map(ob => ({ ob, streak: streaks[ob.id] }))
   386→                    .sort((a, b) => (b.streak?.currentStreak || 0) - (a.streak?.currentStreak || 0))
   387→                    .map(({ ob, streak }, idx) => (
   388→                      <TableRow
   389→                        key={ob.id}
   390→                        className={`${idx % 2 === 0 ? 'data-table-row-even' : 'data-table-row-odd'} table-row-hover-effect`}
   391→                      >
   392→                        <TableCell className="text-sm">
   393→                          <span className={`inline-flex h-6 w-6 items-center justify-center rounded-full text-xs font-bold shrink-0 ${
   394→                            idx === 0 ? 'bg-amber-200 text-amber-700 dark:bg-amber-700 dark:text-amber-300'
   395→                            : idx === 1 ? 'bg-amber-100 text-amber-600 dark:bg-amber-800 dark:text-amber-300'
   396→                            : idx === 2 ? 'bg-gray-200 text-gray-700 dark:bg-gray-600 dark:text-gray-300'
   397→                            : 'bg-muted text-muted-foreground'
   398→                          }`}>
   399→                            {idx + 1}
   400→                          </span>
   401→                        </TableCell>
   402→                        <TableCell>
   403→                          <div>
   404→                            <p className="text-sm font-medium">{ob.name}</p>
   405→                            <p className="text-[11px] text-muted-foreground">{ob.phone || ob.username}</p>
   406→                          </div>
   407→                        </TableCell>
   408→                        <TableCell className="text-center">
   409→                          <StreakBadge streak={streak?.currentStreak || 0} />
   410→                        </TableCell>
   411→                        <TableCell className="text-center hidden sm:table-cell">
   412→                          <div className="flex items-center justify-center gap-1.5">
   413→                            <Trophy className="h-3 w-3 text-slate-500 dark:text-slate-400" />
   414→                            <span className="text-sm font-semibold tabular-nums">{streak?.longestStreak || 0}d</span>
   415→                          </div>
   416→                        </TableCell>
   417→                        <TableCell className="text-center hidden md:table-cell">

   433→            </Table>
   434→          </div>
   435→        </CardContent>
   436→      </Card>
   437→
   438→      {/* Recent Visits */}
   439→      <Card className="card-elevated">
   440→        <CardHeader className="pb-3 pt-4 px-5">
   441→          <div className="flex items-center justify-between gap-4 flex-wrap">
   442→            <CardTitle className="text-base font-semibold flex items-center gap-2">
   443→              <MapPin className="h-4 w-4 text-primary" />
   444→              Recent Shop Visits
   445→              <Badge variant="secondary" className="text-[11px] ml-1">
   446→                {filteredVisits.length} records
   447→              </Badge>
   448→            </CardTitle>
   449→            <div className="flex items-center gap-3">
   450→              <Select value={selectedOB} onValueChange={setSelectedOB}>
   451→                <SelectTrigger className="w-[150px] h-9 text-sm">
   452→                  <SelectValue placeholder="All OBs" />

   456→                  {orderbookers.map(ob => (
   457→                    <SelectItem key={ob.id} value={ob.id}>{ob.name}</SelectItem>
   458→                  ))}
   459→                </SelectContent>
   460→              </Select>
   461→              <div className="relative w-56">
   462→                <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   463→                <Input
   464→                  placeholder="Search visits..."
   465→                  value={searchQuery}
   466→                  onChange={(e) => setSearchQuery(e.target.value)}
   467→                  className="pl-9 h-9 text-sm"
   468→                />
   469→              </div>
   470→            </div>
   471→          </div>
   472→        </CardHeader>
   473→        <CardContent className="p-0">
   474→          {visitsLoading ? (

   476→              <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />

   478→          ) : (
   479→            <div className="overflow-x-auto">
   480→              <Table>
   481→                <TableHeader>
   482→                  <TableRow className="bg-primary hover:bg-transparent">
   483→                    <TableHead className="text-white font-semibold text-xs">Shop</TableHead>
   484→                    <TableHead className="text-white font-semibold text-xs">Orderbooker</TableHead>
   485→                    <TableHead className="text-white font-semibold text-xs text-center">Source</TableHead>
   486→                    <TableHead className="text-white font-semibold text-xs text-right hidden sm:table-cell">Amount</TableHead>
